cmake_minimum_required(VERSION 3.20)
project(mmhk_1d LANGUAGES C CXX Fortran)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(OpenMP REQUIRED)
find_package(Boost REQUIRED)
find_path(LAPACKE_INCLUDE_DIR lapacke.h REQUIRED)
find_library(OPENBLAS_LIBRARY openblas REQUIRED)
find_library(LAPACKE_LIBRARY lapacke REQUIRED)
include(FetchContent)
set(ICB ON CACHE BOOL "ARPACK C binding" FORCE)
set(INTERFACE64 OFF CACHE BOOL "LP64" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static dependency" FORCE)
set(TESTS OFF CACHE BOOL "ARPACK upstream tests" FORCE)
set(BLA_VENDOR OpenBLAS)
FetchContent_Declare(arpackng
 URL https://codeload.github.com/opencollab/arpack-ng/tar.gz/refs/tags/3.9.1
 URL_HASH SHA256=f6641deb07fa69165b7815de9008af3ea47eb39b2bb97521fbf74c97aba6e844
 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(arpackng)
FetchContent_Declare(qbasis
 URL https://codeload.github.com/wztzjhn/quantum_basis/tar.gz/33324ce6081145ea9af981c64b2a2cc32c7faf0a
 URL_HASH SHA256=fbc653c4ed6b05456506d25be7f784db641f08d845923b22ffb00e6feecc5710
 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_GetProperties(qbasis)
if(NOT qbasis_POPULATED)
 FetchContent_Populate(qbasis)
endif()
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/cmake/patch_qbasis.py ${qbasis_SOURCE_DIR} COMMAND_ERROR_IS_FATAL ANY)
set(QB_SOURCES basis ckpt lanczos lattice miscellaneous model operators sparse)
list(TRANSFORM QB_SOURCES PREPEND "${qbasis_SOURCE_DIR}/src/")
list(TRANSFORM QB_SOURCES APPEND ".cc")
add_library(mmhk_qbasis STATIC ${QB_SOURCES})
target_include_directories(mmhk_qbasis PUBLIC reproducibility/compat ${qbasis_SOURCE_DIR}/src ${Boost_INCLUDE_DIRS} ${LAPACKE_INCLUDE_DIR})
target_compile_options(mmhk_qbasis PUBLIC -UNDEBUG)
target_compile_definitions(mmhk_qbasis PUBLIC LAPACK_COMPLEX_CPP)
target_link_libraries(mmhk_qbasis PUBLIC OpenMP::OpenMP_CXX arpack ${LAPACKE_LIBRARY} ${OPENBLAS_LIBRARY})
foreach(kind response spectral)
 add_executable(chain_Hubbard_tbc_${kind} create_code/chain_Hubbard_tbc_${kind}.cc)
 target_link_libraries(chain_Hubbard_tbc_${kind} PRIVATE mmhk_qbasis)
endforeach()
install(TARGETS chain_Hubbard_tbc_response chain_Hubbard_tbc_spectral RUNTIME DESTINATION bin)
